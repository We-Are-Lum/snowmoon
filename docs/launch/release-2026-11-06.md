-- closed-model remix's front view, units 3), counted as the daily limit counts them; a Bluesky account's remix view

declare
  open_remix constant uuid := gen_random_uuid();
  closed_remix constant uuid := gen_random_uuid();
  bluesky_remix constant uuid := gen_random_uuid();

  used int;
  refused boolean;
  bad record;
begin
  insert into studio.sheet_remix_asks (remix, fid, units, slot) values

    (closed_remix, 2147483646, 3, 'front');

  -- Remixes started today, in units: one per remix, its units counted once (1 + 3).

  if used <> 4 then raise exception 'READ-BACK FAILED: the day''s remixes count % units, not 4', used; end if;

      (0::bigint, 1::smallint, 'front', 'zero as a person number'),

      (-999999999::bigint, 1::smallint, 'front', 'the lowest wallet number')

    begin
      insert into studio.sheet_remix_asks (remix, fid, units, slot) values (open_remix, bad.fid, bad.units, bad.slot);

    end;
    if not refused then raise exception 'READ-BACK FAILED: % was accepted', bad.why; end if;
