### 1D. Read-back (b)

Expected: the notice `READ-BACK PASSED`.

```sql
-- Read-back (b): a realistic decision per network goes in as the site writes it, is read back, replaced, refused when
-- malformed, and deleted. Ends with "READ-BACK PASSED", or stops with "READ-BACK FAILED: …"; either way nothing is kept.
do $$
declare
  did constant text := 'did:plc:readback' || repeat('z', 2000);
  lens constant text := '0x00000000000000000000000000000000000c0de1';
  n int;
  refused boolean;
  bad record;
begin
  insert into studio.check_decisions (network, account, threshold, passed) values
    ('farcaster', '987654321', 'neynar-score>=0.7', true),
    ('lens', lens, 'lens-score>=1000', false),
    ('bluesky', did, 'age30d+5posts30d', true);
  select count(*) into n from studio.check_decisions
    where (network, account, threshold) in (('farcaster', '987654321', 'neynar-score>=0.7'), ('lens', lens, 'lens-score>=1000'), ('bluesky', did, 'age30d+5posts30d'))
      and decided_at > now() - interval '1 minute';
  if n <> 3 then raise exception 'READ-BACK FAILED: % of 3 decisions read back', n; end if;
  -- A new decision under the same rule replaces the old one (decide(), src/lib/checks/decisions.ts).
  insert into studio.check_decisions (network, account, threshold, passed) values ('lens', lens, 'lens-score>=1000', true)
    on conflict (network, account, threshold) do update set passed = excluded.passed, decided_at = now();
  select count(*) into n from studio.check_decisions where network = 'lens' and account = lens and passed;
  if n <> 1 then raise exception 'READ-BACK FAILED: the replaced Lens decision reads back % rows', n; end if;
  -- Malformed rows are refused, each by its own check.
  for bad in select * from (values
      ('farcaster', 'abc', 'neynar-score>=0.7', 'a FID that is not digits'),
      ('lens', '0x00000000000000000000000000000000000C0DE1', 'lens-score>=1000', 'a Lens address in capitals'),
      ('bluesky', 'readback.bsky.social', 'age30d+5posts30d', 'a handle in place of a DID'),
      ('mastodon', '987654321', 'neynar-score>=0.7', 'a network other than farcaster, lens or bluesky'),
      ('farcaster', '987654321', 'Score 0.83', 'a rule that is a score, not plain rule text')
    ) as b(network, account, threshold, why) loop
    refused := false;
    begin
      insert into studio.check_decisions (network, account, threshold, passed) values (bad.network, bad.account, bad.threshold, true);
    exception when check_violation then refused := true;
    end;
    if not refused then raise exception 'READ-BACK FAILED: % was accepted', bad.why; end if;
  end loop;
  -- The same account and rule twice, without replacing, is refused (the primary key).
  refused := false;
  begin
    insert into studio.check_decisions (network, account, threshold, passed) values ('farcaster', '987654321', 'neynar-score>=0.7', false);
  exception when unique_violation then refused := true;
  end;
  if not refused then raise exception 'READ-BACK FAILED: a second decision for one account and rule was accepted'; end if;
  delete from studio.check_decisions where account in ('987654321', lens, did);
  select count(*) into n from studio.check_decisions where account in ('987654321', lens, did);
  if n <> 0 then raise exception 'READ-BACK FAILED: % rows left after the delete', n; end if;
  raise notice 'READ-BACK PASSED';
end $$;
```
